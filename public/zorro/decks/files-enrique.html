agent/judge.pyThe program that runs
1"""The judgment half, run headless for testing. In the room it is Claude Code reading ZORRO_AGENT.md;
2here it is `claude -p` with the same files, so we can score it before the week. Subscription billed.
3
4 python3 agent/judge.py 2026-08-31
5"""
6import json, os, subprocess, sys
7from datetime import date
8HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
9asof = sys.argv[1] if len(sys.argv) > 1 else '2026-08-31'
10log = open(os.path.join(HERE, 'out', 'carlos-run.log')).read() if os.path.exists(os.path.join(HERE, 'out', 'carlos-run.log')) else ''
11if f'{asof} rows=' not in log: raise SystemExit(f'Carlos has no log line for {asof}; not opening his file')
12cand = json.load(open(os.path.join(HERE, 'out', f'candidates-{asof}.json')))
13# the model only looks at members who are quieter than their own pattern, or were touched before; the rest are steady
14look = [m for m in cand['members'] if (m['usual_per_week'] is not None and m['usual_per_week'] >= 0.8 and m['visits_last3'] < m['usual_per_week']) or m['previously'] or (m['visits_last3'] == 0 and m['notes']) or (sum(m['visits_per_week_last12'][-2:]) == 0 and (m['usual_per_week'] or 0) >= 0.8)]
15steady = [m for m in cand['members'] if m not in look]
16spec = open(os.path.join(ROOT, 'students', 'enrique-role.md')).read() # the role page IS the spec; same text the students read
17voice = open(os.path.join(ROOT, 'messaging', 'email-voice-and-rules.md')).read()
18goals = open(os.path.join(ROOT, 'messaging', 'messages-by-goal.md')).read()
19prompt = f"""You are Enrique. Today is Monday {asof}. Read your job, the voice and the goals, then decide for every member below.
20
21{spec}
22
23{voice}
24
25{goals}
26
27Members quieter than their own pattern this week (usual_per_week is their average over the nine weeks before the last three; visits_last3 is the last three weeks; visits_per_week_last12 is oldest first):
28{json.dumps(look, ensure_ascii=False)}
29
30Reply with ONLY a JSON array, one object per member above, keys exactly: member_id, decision (draft|skip|ask), reason, email. No prose before or after."""
31r = subprocess.run(['claude', '-p', prompt], stdin=subprocess.DEVNULL, capture_output=True, text=True, timeout=900)
32txt = r.stdout.strip()
33if txt.startswith('```'): txt = txt.split('```')[1].lstrip('json').strip()
34try: dec = json.loads(txt)
35except Exception: open(os.path.join(HERE, 'out', f'judge-raw-{asof}.txt'), 'w').write(r.stdout + '\n---stderr---\n' + r.stderr); raise SystemExit('model did not return JSON; raw saved')
36byid = {d['member_id']: d for d in dec}
37nm = lambda m: f"{m['first_name']} {m['last_name']}"
38rows = [dict(member_id=m['member_id'], name=nm(m), decision='skip', reason='steady against their own pattern', email='') for m in steady]
39for m in look:
40 d = byid.get(m['member_id'], dict(decision='skip', reason='MODEL GAVE NO ANSWER', email=''))
41 rows.append(dict(member_id=m['member_id'], name=nm(m), decision=d.get('decision'), reason=d.get('reason', ''), email=d.get('email', '')))
42json.dump(dict(as_of=asof, decisions=rows), open(os.path.join(HERE, 'out', f'decisions-{asof}.json'), 'w'), indent=1, ensure_ascii=False)
43mem_path = os.path.join(HERE, 'state', 'memory.json')
44memory = json.load(open(mem_path)) if os.path.exists(mem_path) else {}
45for x in rows:
46 if x['decision'] in ('draft', 'ask'):
47 memory[x['member_id']] = dict(name=x['name'], flagged_on=asof, action=x['decision'], reason=x['reason'])
48json.dump(memory, open(mem_path, 'w'), indent=1, ensure_ascii=False)
49from collections import Counter
50print('looked at', len(look), '| decisions:', dict(Counter(x['decision'] for x in rows)))

students/enrique-role.mdHis job page, sent to Claude
1# Enrique. Marta's member-care assistant
2
3The second of the four. Four questions, on one page. Yours will be the same length.
4
5## What Enrique produces
6
7One decision per row of Carlos's file, every Monday, in one shape: member id, decision (draft, skip,
8or ask), the reason in one line quoting the data, and the email or nothing. A draft is an email that
9goes out. Nobody approves it first.
10
11Draft when the member is drifting, which means fewer visits in the last three weeks than they
12usually make in one week, and they joined more than 28 days ago, and their last visit is less than
13eight weeks ago, and previously shows no draft or ask dated within the last 28 days, and neither
14stop below applies.
15
16Ask when the notes or the last reply give a health reason in any wording (injured, injury, physio,
17surgery, doctor, pregnant, recovering, hold off), say the member has moved away, wants to pause or
18cancel, or the reply is angry or says they did not sign up. Ask wins over skip. A goal is not a note.
19
20Skip when the notes or the last reply say the member is elsewhere for a stated time in any wording
21(away, holiday, Galicia, travelling, with family for August), or that the slower pattern is by choice
22(every other week, all he can manage), or when the draft conditions do not hold.
23
24The email is four sentences or fewer, in Marta's voice, to the member by first name, built on the
25messages-by-goal line for the goal on their row, with one ask. Never fees, contracts, renewal or
26discounts.
27
28## Where it lands
29
30Four fields on the member's record in Attio: status (emailed or ask marta), last flagged, reason, the email. One decisions
31file, agent/out/decisions-<YYYY-MM-DD>.json, which Cato reads. And for every draft or ask, one entry
32in agent/state/memory.json keyed on member id: flagged_on (the Monday), action, reason.
33
34## How he knows he is done
35
36The decision rows match Carlos's rows one to one on member id, and a second run on the same Monday
37gives the same decision for every row, because previously ignores entries dated that Monday.
38
39## What he does not do
40
41He never opens the door log, does no arithmetic, and never corrects a desk note. Where a desk note
42does not fit the member's row, the decision is ask and the reason names both.
43
44## The handoff
45
46Carlos's file in, opened only if agent/out/carlos-run.log has a line for this Monday. The record,
47the decisions file and the memory entries out. The two-week check is a separate run that reads the
48returns file and writes the outcome.

messaging/email-voice-and-rules.mdSent to Claude
1# The emails. Marta's voice and the rules the agent follows
2
3The students get this file. They do not have to work out the positioning or the messaging; that
4would eat the week. Paul talks about why the quality of these emails matters. The agent has to
5write to it.
6
7## Who is writing
8
9Marta Ferrer, the owner. She is writing to one person she knows, not to a list. She has noticed
10they have not been in. She is not selling anything and she is not guilt-tripping anyone.
11
12## What every email does
13
14It says who it is from, in the first line, and that Marta noticed. It names the thing the member
15said they wanted, in their own words off the record. It makes one ask, small and specific: a class
16this week, a morning, a reply. It ends. Four or five sentences. Under eighty words.
17
18## What an email never does
19
20It never says "we miss you". It never mentions the membership fee, the contract, or renewal. It
21never asks why they stopped. It never offers a discount. It never lists three things they could do.
22It never uses the word journey. It never guesses at a reason the record does not give.
23
24It never names the member's private life. No wedding, no holiday, no family member, no health
25matter, no date. A gym writing about someone's sister's wedding is intrusive, however well meant.
26The email refers to the goal the member gave when they joined in general words only: "what you
27were working towards", "the goal you set when you joined". Warm, but professional. Paul, 11 Sep 2026.
28
29## When the agent does not write
30
31The record says away, on holiday, travelling. The record says injured, physio, surgery, doctor,
32pregnant. The member joined less than four weeks ago. The member has been gone eight weeks or
33more. The member has always come once a fortnight. A reply has come in that the agent cannot judge.
34In every one of those cases the agent writes nothing and tells Marta why, in one line.
35

38Hi Rubén,

40We have missed you at Gimnasio Zorro! It has been a while since your last visit and we wanted to

47Why it is bad. Nobody wrote it. It does not know who Rubén is or what he wanted. Three asks and a

50## A good one

52Hi Rubén,
53
54Marta here. I noticed you have not been in since the middle of August. You told me when you joined
55what you were working towards, and a few weeks off makes that harder. Tuesday at 7 there is a small
56class that would suit, and I will be on the desk if you want a word first. Come in this week?
57
58Marta
59
60Why it is good. One person to one person. It reminds him he had a goal without naming his private
61life. One ask with a day and a time. No fee, no guilt, no offer. He can say yes in two words.

messaging/messages-by-goal.mdSent to Claude
1# What Marta says to each kind of member
2
3The goal is on the member's record. The agent uses the line for that goal as the middle of the
4email, in Marta's words, and adds the one ask. The email never names the member's private life (the
5rule is in email-voice-and-rules.md): the goal is referred to in general words, never the event, the
6person or the health matter behind it. Nothing in this file is a script to paste; it is what
7the note has to be about.
8
9Lose weight. Do not mention weight. Talk about the routine they had built and the mornings or
10evenings they used to come. The ask is a specific slot they used to keep.
11
12Build muscle. They care about not losing what they built. Say the floor has been quiet in the
13evenings and their bench is free. The ask is a session this week, not a programme.
14
15General fitness. These are the easiest to lose because nothing is pulling them in. Name the class
16or the time they used to come and say it is still there. The ask is one visit.
17
18Stress and sleep. They joined because life was heavy. If they have stopped, life probably got
19heavier. Keep it gentle, no push. The ask is a quiet morning or a stretch class, and it is fine to
20say no.
21
22Get fit for an event. The event is on the record with a date, and the email never names it, never
23names who it is for, and never counts the weeks to it. Say they set themselves a goal with a date
24on it when they joined, and that there is still time to make a difference. The ask is a plan for the
25weeks left, and Marta offering ten minutes at the desk to write it.
26
27Strength as I get older. Usually a member over sixty who was coming steadily. Say plainly that a
28few weeks off costs more at their age and comes back slower, because that is true and they know
29it. The ask is a morning this week.
30
31Rehab after injury. Check the record first. If the notes say the injury is back, do not write.
32If the record is silent, the note asks how the recovery is going and offers a slot with the
33instructor who knows their case. The ask is a reply, not a visit.

agent/monday.pyThe runner the clock starts. It starts all four agents in order.
1"""The runner. This is what the clock starts. Nobody types this on a Monday.
2
3Runs the four agents in order and stops the moment one of them has not left its file:
4 1. Carlos prepare_week.py writes out/candidates-<monday>.json
5 2. Enrique judge.py reads that, writes out/decisions-<monday>.json
6 3. Cato cato.py reads both, writes out/cato-<monday>.json
7 4. Rosa check_returns.py two weeks after a Monday with emails, writes out/returns-<that monday>.json
8Then, if there is an Attio token, the decisions go onto the cards.
9
10Every step writes one line to state/heartbeat.log, and the last line for a Monday says done or
11where it stopped. That file is the proof it ran. A scheduler that says "loaded" proves nothing.
12
13The clock ticks every half hour (see schedule.sh). Most ticks do nothing: not a Monday, or this
14Monday is already done. That is on purpose. A laptop with the lid closed at 6am misses a 6am alarm,
15so the agent asks "has this week's Monday been done?" every time it wakes, and does it if not.
16
17 python3 agent/monday.py the most recent Monday, once
18 python3 agent/monday.py --monday 2026-09-07 a named Monday
19 python3 agent/monday.py --force run it again even if the heartbeat says done
20 python3 agent/monday.py --no-attio leave the cards alone (tests)
21"""
22import os, subprocess, sys
23from datetime import date, datetime, timedelta
24
25HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
26PY = sys.executable
27HEART = os.path.join(HERE, 'state', 'heartbeat.log')
28args = sys.argv[1:]
29force = '--force' in args
30no_attio = '--no-attio' in args or os.environ.get('ZORRO_NO_ATTIO') == '1'
31if '--monday' in args:
32 monday = date.fromisoformat(args[args.index('--monday') + 1])
33else:
34 today = date.today(); monday = today - timedelta(days=today.weekday())
35assert monday.weekday() == 0, 'the agent runs as of a Monday'
36M = monday.isoformat()
37
38def beat(text):
39 os.makedirs(os.path.dirname(HEART), exist_ok=True)
40 line = f'{datetime.now().strftime("%Y-%m-%d %H:%M:%S")} monday={M} {text}'
41 open(HEART, 'a').write(line + '\n'); print(line)
42
43def already_done():
44 """Done means the heartbeat says so, or the three Monday files are already there (a run from before the heartbeat existed)."""
45 if os.path.exists(HEART) and any(f'monday={M} done' in l for l in open(HEART).read().split('\n')): return True
46 return all(os.path.exists(os.path.join(HERE, 'out', f'{n}-{M}.json')) for n in ('candidates', 'decisions', 'cato'))
47
48def run(name, script, expect, extra=()):
49 """Run one agent. It is done only when the file it promised exists. Otherwise stop everything."""
50 r = subprocess.run([PY, os.path.join(HERE, script), *extra], cwd=ROOT, stdin=subprocess.DEVNULL,
51 capture_output=True, text=True, timeout=1800)
52 ok = r.returncode == 0 and os.path.exists(os.path.join(ROOT, expect))
53 tail = (r.stdout.strip().split('\n') or [''])[-1][:160] if r.stdout.strip() else (r.stderr.strip().split('\n') or [''])[-1][:160]
54 beat(f'{name} {"ok" if ok else "FAILED"} {tail}')
55 if not ok:
56 beat(f'STOPPED at {name}: {expect} was not written. Nothing after it ran.')
57 sys.exit(1)
58
59if already_done() and not force:
60 print(f'{M} is already done in {os.path.relpath(HEART, ROOT)}; nothing to do'); sys.exit(0)
61
62beat('start')
63run('carlos', 'prepare_week.py', f'agent/out/candidates-{M}.json', (M,))
64run('enrique', 'judge.py', f'agent/out/decisions-{M}.json', (M,))
65run('cato', 'cato.py', f'agent/out/cato-{M}.json', (M,))
66
67# Rosa looks back two weeks. She runs only when there was a Monday with emails then and she has not checked it yet.
68import json
69back = (monday - timedelta(days=14)).isoformat()
70memory = json.load(open(os.path.join(HERE, 'state', 'memory.json'))) if os.path.exists(os.path.join(HERE, 'state', 'memory.json')) else {}
71wrote_then = any(v.get('flagged_on') == back and v.get('action') == 'draft' for v in memory.values())
72if wrote_then and not os.path.exists(os.path.join(HERE, 'out', f'returns-{back}.json')):
73 run('rosa', 'check_returns.py', f'agent/out/returns-{back}.json', (back,))
74elif wrote_then:
75 beat(f'rosa skipped: returns-{back}.json already exists')
76else:
77 beat(f'rosa skipped: nothing was emailed on {back}')
78
79def attio(mode, when):
80 """attio.py decisions <monday> puts Enrique's reason and email on the cards; attio.py returns <monday> moves Rosa's."""
81 r = subprocess.run([PY, os.path.join(HERE, 'attio.py'), mode, when], cwd=ROOT, stdin=subprocess.DEVNULL, capture_output=True, text=True, timeout=1800)
82 tail = (r.stdout.strip() or r.stderr.strip()).split('\n')[-1][:160]
83 beat(f'attio {mode} {"ok" if r.returncode == 0 else "FAILED"} {tail}')
84 if r.returncode != 0:
85 beat(f'STOPPED at attio {mode}: the cards were not written.'); sys.exit(1)
86
87if no_attio:
88 beat('attio skipped: --no-attio')
89else:
90 attio('decisions', M)
91 attio('cato', M) # Cato's line on every card where he disagrees (17 Sep: the code existed, the runner never called it)
92 if wrote_then: attio('returns', back)
93
94beat('done')
