agent/prepare_week.pyThe program that runs
1"""The reading half of the Zorro agent. Deterministic, no model.
2
3Given a Monday, it reads the door log and the member file and writes agent/out/candidates-<date>.json:
4for every member, their usual visits per week (the nine weeks before the last three), their last three
5weeks, whether they are new, whether they have been gone a long time, their notes, and what the agent
6did for them before (from agent/state/memory.json). The judgment and the email are the model's job,
7in Claude Code, reading this file and the messaging pack. Nothing here decides who gets written to.
8
9 python3 agent/prepare_week.py 2026-08-31
10"""
11import csv, json, os, sys
12from collections import Counter, defaultdict
13from datetime import datetime, date, timedelta
14
15HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
16asof = date.fromisoformat(sys.argv[1]) if len(sys.argv) > 1 else date(2026, 8, 31)
17assert asof.weekday() == 0, 'the agent runs on a Monday'
18mem = {r['member_id']: r for r in csv.DictReader(open(os.path.join(ROOT, 'data', 'members.csv')))}
19visits = defaultdict(list)
20for r in csv.DictReader(open(os.path.join(ROOT, 'data', 'checkins.csv'))):
21 d = datetime.strptime(r['checkin_at'], '%Y-%m-%d %H:%M').date()
22 if d < asof: visits[r['member_id']].append(d)
23LOG = os.path.join(HERE, 'out', 'carlos-run.log')
24def log_line(text):
25 lines = [l for l in open(LOG).read().split('\n') if l and not l.startswith(asof.isoformat())] if os.path.exists(LOG) else []
26 open(LOG, 'w').write('\n'.join(lines + [text]) + '\n')
27for need in ('data/checkins.csv', 'data/members.csv', 'data/email_log.csv', 'agent/state/memory.json'):
28 if not os.path.exists(os.path.join(ROOT, need)):
29 log_line(f'{asof} NO FILE WRITTEN: missing {need}'); raise SystemExit(f'missing {need}; no candidates file written')
30memory = json.load(open(os.path.join(HERE, 'state', 'memory.json')))
31last_sent = {}
32replies = {}
33lp = os.path.join(ROOT, 'data', 'email_log.csv')
34if os.path.exists(lp):
35 for r in csv.DictReader(open(lp)):
36 sd = r['sent_at'][:10]
37 if date.fromisoformat(sd) < asof:
38 last_sent[r['member_id']] = max(last_sent.get(r['member_id'], ''), sd)
39 if r['replied'] == 'yes' and sd >= replies.get(r['member_id'], {}).get('sent_date', ''): replies[r['member_id']] = dict(text=r['reply_text'], sent_date=sd)

41def weeks_back(dates, start, n):
42 """visits per week for the n weeks ending the day before `start`, oldest first"""
43 out = []
44 for i in range(n, 0, -1):
45 a = start - timedelta(days=7 * i); b = a + timedelta(days=7)
46 out.append(sum(1 for d in dates if a <= d < b))
47 return out
48
49rows = []
50for m, r in mem.items():
51 last12 = weeks_back(visits[m], asof, 12)
52 joined = date.fromisoformat(r['joined'])
53 days_since = (asof - joined).days
54 gaps = []
55 before = sum(1 for d in visits[m] if d < joined)
56 if before: gaps.append(f'{before} swipe(s) dated before joining, not counted')
57 kept = [d for d in visits[m] if d >= joined]
58 last12 = weeks_back(kept, asof, 12)
59 first9_start = asof - timedelta(days=7 * 12)
60 if joined <= first9_start:
61 usual = round(sum(last12[:9]) / 9.0, 2)
62 else:
63 full_weeks = min(9, max(0, (asof - timedelta(days=21) - joined).days // 7))
64 usual = round(sum(last12[:9]) / full_weeks, 2) if full_weeks else None
65 if usual is None: gaps.append('usual rate: no full week between joining and the last three weeks')
66 recent = sum(last12[9:])
67 rows.append(dict(member_id=m, first_name=r['first_name'], last_name=r['last_name'], email=r['email'], joined=r['joined'],
68 weeks_since_joined=days_since // 7, days_since_joined=days_since, plan=r['plan'], goal=r['goal'], notes=r['notes'],
69 visits_per_week_last12=last12, usual_per_week=usual, visits_last3=recent,
70 weeks_since_last_visit=(asof - (max(kept) if kept else joined)).days // 7,
71 previously=(memory.get(m) if memory.get(m) and memory[m].get('flagged_on','') < asof.isoformat() else None), last_reply=replies.get(m), last_sent=last_sent.get(m), gaps='; '.join(gaps)))
72out = os.path.join(HERE, 'out', f'candidates-{asof}.json')
73json.dump(dict(as_of=asof.isoformat(), members=rows), open(out, 'w'), indent=1, ensure_ascii=False)
74quiet = [x for x in rows if x['usual_per_week'] is not None and x['usual_per_week'] >= 0.8 and x['visits_last3'] < x['usual_per_week']]
75log_line(f'{asof} rows={len(rows)} file={os.path.relpath(out, ROOT)}')
76print(f'as of {asof}: {len(rows)} members read, {len(quiet)} quieter than their own pattern (before judgment), written to {os.path.relpath(out, ROOT)}')

students/carlos-role.mdThe page the program was written from
1# Carlos. The one who pulls the numbers and cannot write
2
3The first of the four. Four questions, on one page. His code turns four files into one file Enrique
4can judge from, the same shape every Monday. He is not allowed to write a word to a member or to
5describe a number. In real agent teams the mistakes are almost never in the fetching, they are in the
6sentence written next to what was fetched.
7
8## What Carlos produces
9
10One file per Monday, agent/out/candidates-<YYYY-MM-DD>.json, one row per member id in the member
11file, with exactly these keys: member_id, first_name, last_name, email, plan, goal, notes, joined,
12days_since_joined, weeks_since_joined, visits_per_week_last12 (twelve whole numbers, oldest week
13first), usual_per_week, visits_last3, weeks_since_last_visit, previously, last_reply, last_sent, gaps.
14
15A week runs Monday to Sunday and the twelve weeks are the twelve full weeks ending the Sunday before
16the Monday being run. Only swipes dated before the Monday and on or after the joined date are
17counted; a swipe before joining is not counted and gaps says how many. usual_per_week is visits in
18the first nine of the twelve weeks divided by nine, to two decimal places; for a member who joined
19after the start of those nine weeks it is visits in the full weeks between joining and the last
20three weeks divided by that number of weeks, and empty with a gaps line if there is none.
21weeks_since_last_visit is days from the last counted swipe to the Monday divided by seven, whole
22weeks; a member with no counted swipe counts as last visited on the day they joined. previously is
23the member's entry in agent/state/memory.json copied as is (flagged_on, action, reason), only if its
24flagged_on is before the Monday; entries dated the Monday itself are ignored, so a rerun after Enrique
25has run still gives the same file. days_since_joined is the Monday minus the joined date;
26weeks_since_joined is that divided by seven, whole weeks; visits_last3 is the sum of the last three of
27the twelve weekly counts.
28last_reply is the reply_text of the email-log row with replied = yes and the latest sent_at before
29the Monday, with that sent_at date. last_sent is the latest sent_at before the Monday, or empty.
30
31## Where it lands
32
33That file, and one line for that Monday in agent/out/carlos-run.log: the Monday, the row count, the
34path. A rerun of the same Monday replaces its line.
35
36## How he knows he is done
37
38Every row in the member file has a row, every number can be recomputed from the four files by anyone
39who checks, and two runs on the same Monday from the same four files give the same file byte for
40byte. If any of the four files is missing he writes no file and one log line naming the path.
41
42## What he does not do
43
44He does not judge who is drifting, write notes, touch Attio, round beyond two places, estimate, say
45"about", or filter: every member gets a row, whatever their plan or status.
46
47## The handoff
48
49In: the door log, the member file, the email log, the memory file. Out: the file and the log line.
50Enrique may not go back to the door log, so if this file is wrong every decision is wrong, and the
51only place to fix it is here.
52
agent/monday.pyThe runner the clock starts. It starts all four agents in order.
1"""The runner. This is what the clock starts. Nobody types this on a Monday.
2
3Runs the four agents in order and stops the moment one of them has not left its file:
4 1. Carlos prepare_week.py writes out/candidates-<monday>.json
5 2. Enrique judge.py reads that, writes out/decisions-<monday>.json
6 3. Cato cato.py reads both, writes out/cato-<monday>.json
7 4. Rosa check_returns.py two weeks after a Monday with emails, writes out/returns-<that monday>.json
8Then, if there is an Attio token, the decisions go onto the cards.
9
10Every step writes one line to state/heartbeat.log, and the last line for a Monday says done or
11where it stopped. That file is the proof it ran. A scheduler that says "loaded" proves nothing.
12
13The clock ticks every half hour (see schedule.sh). Most ticks do nothing: not a Monday, or this
14Monday is already done. That is on purpose. A laptop with the lid closed at 6am misses a 6am alarm,
15so the agent asks "has this week's Monday been done?" every time it wakes, and does it if not.
16
17 python3 agent/monday.py the most recent Monday, once
18 python3 agent/monday.py --monday 2026-09-07 a named Monday
19 python3 agent/monday.py --force run it again even if the heartbeat says done
20 python3 agent/monday.py --no-attio leave the cards alone (tests)
21"""
22import os, subprocess, sys
23from datetime import date, datetime, timedelta
24
25HERE = os.path.dirname(os.path.abspath(__file__)); ROOT = os.path.dirname(HERE)
26PY = sys.executable
27HEART = os.path.join(HERE, 'state', 'heartbeat.log')
28args = sys.argv[1:]
29force = '--force' in args
30no_attio = '--no-attio' in args or os.environ.get('ZORRO_NO_ATTIO') == '1'
31if '--monday' in args:
32 monday = date.fromisoformat(args[args.index('--monday') + 1])
33else:
34 today = date.today(); monday = today - timedelta(days=today.weekday())
35assert monday.weekday() == 0, 'the agent runs as of a Monday'
36M = monday.isoformat()
37
38def beat(text):
39 os.makedirs(os.path.dirname(HEART), exist_ok=True)
40 line = f'{datetime.now().strftime("%Y-%m-%d %H:%M:%S")} monday={M} {text}'
41 open(HEART, 'a').write(line + '\n'); print(line)
42
43def already_done():
44 """Done means the heartbeat says so, or the three Monday files are already there (a run from before the heartbeat existed)."""
45 if os.path.exists(HEART) and any(f'monday={M} done' in l for l in open(HEART).read().split('\n')): return True
46 return all(os.path.exists(os.path.join(HERE, 'out', f'{n}-{M}.json')) for n in ('candidates', 'decisions', 'cato'))
47
48def run(name, script, expect, extra=()):
49 """Run one agent. It is done only when the file it promised exists. Otherwise stop everything."""
50 r = subprocess.run([PY, os.path.join(HERE, script), *extra], cwd=ROOT, stdin=subprocess.DEVNULL,
51 capture_output=True, text=True, timeout=1800)
52 ok = r.returncode == 0 and os.path.exists(os.path.join(ROOT, expect))
53 tail = (r.stdout.strip().split('\n') or [''])[-1][:160] if r.stdout.strip() else (r.stderr.strip().split('\n') or [''])[-1][:160]
54 beat(f'{name} {"ok" if ok else "FAILED"} {tail}')
55 if not ok:
56 beat(f'STOPPED at {name}: {expect} was not written. Nothing after it ran.')
57 sys.exit(1)
58
59if already_done() and not force:
60 print(f'{M} is already done in {os.path.relpath(HEART, ROOT)}; nothing to do'); sys.exit(0)
61
62beat('start')
63run('carlos', 'prepare_week.py', f'agent/out/candidates-{M}.json', (M,))
64run('enrique', 'judge.py', f'agent/out/decisions-{M}.json', (M,))
65run('cato', 'cato.py', f'agent/out/cato-{M}.json', (M,))
66
67# Rosa looks back two weeks. She runs only when there was a Monday with emails then and she has not checked it yet.
68import json
69back = (monday - timedelta(days=14)).isoformat()
70memory = json.load(open(os.path.join(HERE, 'state', 'memory.json'))) if os.path.exists(os.path.join(HERE, 'state', 'memory.json')) else {}
71wrote_then = any(v.get('flagged_on') == back and v.get('action') == 'draft' for v in memory.values())
72if wrote_then and not os.path.exists(os.path.join(HERE, 'out', f'returns-{back}.json')):
73 run('rosa', 'check_returns.py', f'agent/out/returns-{back}.json', (back,))
74elif wrote_then:
75 beat(f'rosa skipped: returns-{back}.json already exists')
76else:
77 beat(f'rosa skipped: nothing was emailed on {back}')
78
79def attio(mode, when):
80 """attio.py decisions <monday> puts Enrique's reason and email on the cards; attio.py returns <monday> moves Rosa's."""
81 r = subprocess.run([PY, os.path.join(HERE, 'attio.py'), mode, when], cwd=ROOT, stdin=subprocess.DEVNULL, capture_output=True, text=True, timeout=1800)
82 tail = (r.stdout.strip() or r.stderr.strip()).split('\n')[-1][:160]
83 beat(f'attio {mode} {"ok" if r.returncode == 0 else "FAILED"} {tail}')
84 if r.returncode != 0:
85 beat(f'STOPPED at attio {mode}: the cards were not written.'); sys.exit(1)
86
87if no_attio:
88 beat('attio skipped: --no-attio')
89else:
90 attio('decisions', M)
91 attio('cato', M) # Cato's line on every card where he disagrees (17 Sep: the code existed, the runner never called it)
92 if wrote_then: attio('returns', back)
93
94beat('done')
